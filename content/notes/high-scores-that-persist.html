Difficulty: ⭐⭐ moderate · Payoff: ⭐⭐⭐ huge (on Friday) · Time: 30 minutes

A high score that survives closing the game. Then a leaderboard with names on it.

This is the feature that turns showcase night into an actual arcade. A game with a high score table gets played over and over by the same people; a game without one gets played once. Arcade Architects, after all.

#Where saved files go

"user://highscore.save"

user:// is a real folder Godot manages per project (%APPDATA%\Godot\app_userdata\YourGame on Windows). Use it for anything you write at runtime.

⚠️Never save to res://

res:// is your project folder. It's read-only in an exported game, so saving there works while you're testing in the editor and silently fails for everyone at the showcase. Always user://.

#The simplest version: one number

# in your Global autoload -- see Autoload and Global
const SAVE_PATH = "user://highscore.save"

var high_score = 0

func _ready():
	load_high_score()

func load_high_score():
	if not FileAccess.file_exists(SAVE_PATH):
		return                      # first ever run -- leave it at 0
	var f = FileAccess.open(SAVE_PATH, FileAccess.READ)
	high_score = f.get_var()
	f.close()

func save_high_score():
	var f = FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	f.store_var(high_score)
	f.close()

func submit_score(score):
	if score > high_score:
		high_score = score
		save_high_score()
		return true                 # tell the caller it was a new record
	return false

Use it:

func game_over():
	var is_record = Global.submit_score(Global.score)
	$UI/FinalScore.text = "Score: %d" % Global.score
	$UI/HighScore.text  = "Best: %d"  % Global.high_score
	if is_record:
		$UI/NewRecord.show()        # make a fuss -- this is the fun part

The file_exists check is the whole trick: the first time anyone plays there's no file, and reading a missing file crashes.

#A real leaderboard: top 10 with names

# global.gd
const SAVE_PATH = "user://scores.json"
const MAX_ENTRIES = 10

var scores = []      # [{"name": "ABC", "score": 1200}, ...]

func _ready():
	load_scores()

func load_scores():
	if not FileAccess.file_exists(SAVE_PATH):
		return
	var f = FileAccess.open(SAVE_PATH, FileAccess.READ)
	var text = f.get_as_text()
	f.close()
	var data = JSON.parse_string(text)
	if data is Array:
		scores = data               # guard against a corrupt/empty file

func save_scores():
	var f = FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	f.store_string(JSON.stringify(scores))
	f.close()

func add_score(player_name, score):
	scores.append({"name": player_name, "score": score})
	scores.sort_custom(func(a, b): return a["score"] > b["score"])
	if scores.size() > MAX_ENTRIES:
		scores.resize(MAX_ENTRIES)
	save_scores()

func makes_leaderboard(score):
	return scores.size() < MAX_ENTRIES or score > scores[-1]["score"]

JSON is just text, so you can open scores.json in Notepad and read it — very handy for debugging, and for resetting the board between showcase sessions.

sort_custom with func(a, b): return a["score"] > b["score"] is a lambda — a function written inline. It means "a comes before b when a's score is higher."

#Displaying the board

func show_leaderboard():
	var text = ""
	for i in Global.scores.size():
		var entry = Global.scores[i]
		text += "%d. %s  %d\n" % [i + 1, entry["name"], entry["score"]]
	$UI/LeaderboardLabel.text = text

Use a monospace font on that Label so the columns line up (Label and CanvasLayer).

#Three-letter initials (the arcade way)

Don't use a keyboard text box. Three letters you cycle with up/down is the authentic arcade feel and it's more fun:

extends Control

var letters = ["A", "A", "A"]
var current = 0

func _ready():
	update_display()

func _unhandled_input(event):
	if event.is_action_pressed("move_up"):
		cycle(1)
	elif event.is_action_pressed("move_down"):
		cycle(-1)
	elif event.is_action_pressed("move_right"):
		current = min(current + 1, 2)
	elif event.is_action_pressed("move_left"):
		current = max(current - 1, 0)
	elif event.is_action_pressed("ui_accept"):
		submit()
	update_display()

func cycle(direction):
	var code = letters[current].unicode_at(0) - 65     # A = 0
	code = wrapi(code + direction, 0, 26)
	letters[current] = char(65 + code)

func update_display():
	$Initials.text = "".join(letters)

func submit():
	Global.add_score("".join(letters), Global.score)
	get_tree().change_scene_to_file("res://leaderboard.tscn")

wrapi(value, 0, 26) wraps around, so past Z comes back to A.

#Saving other things too

The same file can hold anything — swap the single value for a dictionary:

var save_data = {
	"high_score": 0,
	"levels_unlocked": 1,
	"music_volume": 0.8,
}

Store it with JSON.stringify(save_data) and read it back with JSON.parse_string(). That's a complete save system for a small game. (For anything bigger, see Save Systems.)
